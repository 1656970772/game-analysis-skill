# 模型预览工具 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 GameAnalysis 增加一个固定在 skill `tools/` 目录的本地模型预览 EXE，并让每次导出的 study-root 只生成 BAT 启动器、manifest、缩略图和 FBX 预览缓存。

**Architecture:** 采用“配置 + Python 生成链路 + Electron/Three.js 固定工具”的分层架构。Python 脚本负责配置加载、模型扫描、manifest、BAT、Blender 缩略图/FBX 预转换；Electron 工具只读加载传入的 manifest，通过本地安全资源协议预览模型，不递归扫描任意磁盘。

**Tech Stack:** Python 3 + pytest、PowerShell 5.1+、Electron、Three.js、electron-builder、Blender CLI、Windows BAT。

```python
from __future__ import annotations

import hashlib
import json
from pathlib import Path
from typing import Any


def load_model_viewer_config(config_path: Path) -> dict[str, Any]:
    data = json.loads(config_path.read_text(encoding="utf-8"))
    required = ["tool", "launcher", "formats", "manifest", "paths", "thumbnail", "limits"]
    missing = [key for key in required if key not in data]
    if missing:
        raise ValueError(f"model viewer config missing keys: {', '.join(missing)}")
    return data


def resolve_existing(path: Path) -> Path:
    return path.expanduser().resolve(strict=False)


def ensure_inside_root(path: Path, root: Path) -> Path:
    resolved_path = resolve_existing(path)
    resolved_root = resolve_existing(root)
    try:
        resolved_path.relative_to(resolved_root)
    except ValueError as exc:
        raise ValueError(f"path is outside study-root: {resolved_path}") from exc
    return resolved_path


def stable_model_id(source_path: Path, root: Path) -> str:
    resolved_source = ensure_inside_root(source_path, root)
    relative = resolved_source.relative_to(resolve_existing(root)).as_posix().lower()
    return hashlib.sha256(relative.encode("utf-8")).hexdigest()[:16]


def write_json_utf8(path: Path, payload: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps(payload, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )
```

创建 `scripts/build_model_gallery_manifest.py`，实现以下接口和 CLI。第一版不在 Electron 中直接加载 FBX：FBX 初始 `preview_status` 必须保持 `unsupported`，只有 Task 5 在 Blender 可用并成功导出 `preview-cache/*.preview.glb` 后才改为 `converted`；当 Blender 不可用或未运行时，FBX 继续保持 `unsupported`。

```python
from __future__ import annotations

import argparse
import json
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

from scripts.model_gallery_common import ensure_inside_root, stable_model_id, write_json_utf8


def configured_path(root: Path, config: dict[str, Any], key: str) -> Path:
    return root / Path(config["paths"][key])


def is_external_local_uri(uri: str) -> bool:
    if not uri or uri.startswith("data:"):
        return False
    if "://" in uri or uri.startswith("/") or uri.startswith("\\"):
        return False
    return True


def collect_gltf_external_uris(model_file: Path) -> list[str]:
    if model_file.suffix.lower() != ".gltf":
        return []
    try:
        payload = json.loads(model_file.read_text(encoding="utf-8"))
    except json.JSONDecodeError:
        return []
    uris: list[str] = []
    for buffer in payload.get("buffers", []):
        uri = buffer.get("uri", "")
        if is_external_local_uri(uri):
            uris.append(uri)
    for image in payload.get("images", []):
        uri = image.get("uri", "")
        if is_external_local_uri(uri):
            uris.append(uri)
    return uris


def material_status(config: dict[str, Any], key: str) -> str:
    statuses = config["manifest"]["material_status"]
    if key not in statuses:
        raise ValueError(f"unknown material status: {key}")
    return key


def inspect_gltf_resources(model_file: Path, config: dict[str, Any]) -> tuple[str, str, list[str]]:
    if model_file.suffix.lower() == ".glb":
        return "unknown", material_status(config, "unknown"), []
    if model_file.suffix.lower() != ".gltf":
        return "unknown", material_status(config, "unknown"), []
    text = model_file.read_text(encoding="utf-8", errors="ignore")
    uris = collect_gltf_external_uris(model_file)
    missing = [uri for uri in uris if not (model_file.parent / uri).exists()]
    if missing:
        return "missing", material_status(config, "missing_external_resources"), [f"missing external resource: {uri}" for uri in missing]
    if "baseColorTexture" in text and "\"images\"" in text:
        return "bound", material_status(config, "verified"), []
    if "textures" in text or "\"images\"" in text:
        return "same_name_candidate", material_status(config, "unknown"), []
    return "none", material_status(config, "unknown"), []


def iter_model_files(models_dir: Path, supported_suffixes: list[str]) -> list[Path]:
    suffixes = {f".{suffix.lower().lstrip('.')}" for suffix in supported_suffixes}
    return sorted(
        [path for path in models_dir.rglob("*") if path.is_file() and path.suffix.lower() in suffixes],
        key=lambda path: path.relative_to(models_dir).as_posix().lower(),
    )


def build_item(root: Path, model_file: Path, config: dict[str, Any]) -> dict[str, Any]:
    model_id = stable_model_id(model_file, root)
    source_format = model_file.suffix.lower().lstrip(".")
    texture_status, material_status_value, resource_errors = inspect_gltf_resources(model_file, config)
    preview_status = "ready" if source_format in config["formats"]["direct_preview"] else "unsupported"
    preview_index = configured_path(root, config, "preview_index_dir")
    preview_cache = preview_index / config["paths"]["preview_cache_dir"]
    thumbs = preview_index / config["paths"]["thumbs_dir"]
    preview_path = model_file
    if source_format in config["formats"]["blender_preview"]:
        preview_path = preview_cache / f"{model_id}.preview.glb"
    tags = build_tags(source_format, texture_status, preview_status, material_status_value, config)
    return {
        "id": model_id,
        "name": model_file.stem,
        "source_format": source_format,
        "source_path": str(ensure_inside_root(model_file, root)),
        "preview_path": str(preview_path),
        "thumbnail_path": str(thumbs / f"{model_id}.png"),
        "model_dir": str(model_file.parent.resolve(strict=False)),
        "texture_status": texture_status,
        "material_status": material_status_value,
        "preview_status": preview_status,
        "tags": tags,
        "stats": {"mesh_count": 0, "material_count": 0, "animation_count": 0},
        "errors": resource_errors,
    }


def build_tags(source_format: str, texture_status: str, preview_status: str, material_status: str, config: dict[str, Any]) -> list[str]:
    rules = config["manifest"]["tag_rules"]
    tags: list[str] = []
    if rules["include_source_format"]:
        tags.append(source_format)
    tags.append(f"{rules['texture_prefix']}{texture_status}")
    tags.append(f"{rules['preview_prefix']}{preview_status}")
    tags.append(f"{rules['material_prefix']}{material_status}")
    return tags


def build_manifest(study_root: Path, config: dict[str, Any], manifest_path: Path | None = None) -> dict[str, Any]:

from __future__ import annotations

import argparse
import json
from pathlib import Path

from scripts.model_gallery_common import load_model_viewer_config


SECTION = "## 模型预览"


def replace_section(text: str, section_title: str, replacement: str) -> str:
    start = text.find(section_title)
    if start < 0:
        return text.rstrip() + "\n\n" + replacement.rstrip() + "\n"
    next_start = text.find("\n## ", start + len(section_title))
    if next_start < 0:
        return text[:start].rstrip() + "\n\n" + replacement.rstrip() + "\n"
    return text[:start].rstrip() + "\n\n" + replacement.rstrip() + "\n\n" + text[next_start:].lstrip()


def summarize_manifest(manifest_path: Path) -> tuple[int, int]:
    if not manifest_path.exists():
        return 0, 0
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    items = manifest.get("items", [])
    failed = sum(1 for item in items if item.get("preview_status") == "failed" or item.get("errors"))
    return len(items), failed


def update_summary(study_root: Path, manifest_path: Path, config: dict, viewer_exists: bool, launcher_exists: bool) -> Path:
    report = study_root / Path(config["paths"]["report_dir"]) / config["paths"]["overview_name"]
    report.parent.mkdir(parents=True, exist_ok=True)
    existing = report.read_text(encoding="utf-8") if report.exists() else "# 研究总览\n"
    total, failed = summarize_manifest(manifest_path)
    launcher_status = "已生成" if launcher_exists else "未生成"
    tool_status = "GameModelViewer 可用" if viewer_exists else "GameModelViewer 缺失"
    launcher_path = f"{config['paths']['report_dir']}/{config['launcher']['bat_name']}"
    section = "\n".join([
        SECTION,
        "",
        f"入口：`{launcher_path}`",
        f"模型数量：{total}",
        f"失败数量：{failed}",
        f"启动器状态：{launcher_status}",
        f"工具状态：{tool_status}",
    ])
    report.write_text(replace_section(existing, SECTION, section), encoding="utf-8")
    return report


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--study-root", required=True)
    parser.add_argument("--manifest", required=True)
    parser.add_argument("--viewer-exe", required=True)
    parser.add_argument("--config", default="references/model-viewer-gallery-config.json")
    args = parser.parse_args()
    root = Path(args.study_root)
    config = load_model_viewer_config(Path(args.config))
    launcher = root / Path(config["paths"]["report_dir"]) / config["launcher"]["bat_name"]
    update_summary(root, Path(args.manifest), config, Path(args.viewer_exe).exists(), launcher.exists())
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

from __future__ import annotations

import argparse
import json
import subprocess
from pathlib import Path
from typing import Any

from scripts.model_gallery_common import load_model_viewer_config, write_json_utf8


def plan_blender_jobs(manifest: dict[str, Any], config: dict[str, Any]) -> list[dict[str, Any]]:
    blender_preview = set(config["formats"].get("blender_preview", []))
    jobs = []
    for item in manifest.get("items", []):
        jobs.append({
            "id": item["id"],
            "source_path": item["source_path"],
            "thumbnail_path": item["thumbnail_path"],
            "preview_path": item["preview_path"],
            "export_glb": item["source_format"] in blender_preview,
        })
    return jobs


def append_error_jsonl(errors_path: Path, record: dict[str, Any]) -> None:
    errors_path.parent.mkdir(parents=True, exist_ok=True)
    with errors_path.open("a", encoding="utf-8") as handle:
        handle.write(json.dumps(record, ensure_ascii=False) + "\n")


def update_manifest_after_jobs(manifest_path: Path, results: dict[str, dict[str, Any]], config: dict[str, Any]) -> None:
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    errors_path = manifest_path.parent / config["paths"]["errors_name"]
    errors_path.parent.mkdir(parents=True, exist_ok=True)
    errors_path.touch(exist_ok=True)
    for item in manifest.get("items", []):
        result = results.get(item["id"])
        if not result:
            continue
        if result.get("ok"):
            if item["source_format"] == "fbx":
                item["preview_status"] = "converted"
        else:
            item["preview_status"] = "failed"
            error = str(result.get("error", "thumbnail render failed"))
            item.setdefault("errors", []).append(error)
            append_error_jsonl(errors_path, {
                "id": item["id"],
                "source_path": item.get("source_path", ""),
                "stage": "blender_preview",
                "error": error,
            })
    write_json_utf8(manifest_path, manifest)


def run_blender(blender: Path, worker: Path, job_file: Path, result_file: Path) -> int:
    command = [
        str(blender),
        "--background",
        "--python",
        str(worker),
        "--",
        "--jobs",
        str(job_file),
        "--results",
        str(result_file),
    ]
    return subprocess.run(command, check=False).returncode


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--manifest", required=True)
    parser.add_argument("--blender", required=True)
    parser.add_argument("--config", default="references/model-viewer-gallery-config.json")
    args = parser.parse_args()
    manifest_path = Path(args.manifest)
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    config = load_model_viewer_config(Path(args.config))
    jobs_path = manifest_path.with_name("blender-jobs.json")
    results_path = manifest_path.with_name("blender-results.json")
    jobs = plan_blender_jobs(manifest, config)
    write_json_utf8(jobs_path, {"jobs": jobs, "thumbnail": config["thumbnail"]})
    code = run_blender(Path(args.blender), Path(__file__).with_name("blender_model_preview_worker.py"), jobs_path, results_path)
    if results_path.exists():
        results = json.loads(results_path.read_text(encoding="utf-8"))
    elif code != 0:
        results = {job["id"]: {"ok": False, "error": f"blender exited with code {code}"} for job in jobs}

        results = {}
    update_manifest_after_jobs(manifest_path, results, config)
    return code

def frame_camera(thumbnail: dict) -> None:
    yaw = math.radians(float(thumbnail["camera_yaw_degrees"]))

    distance = 5 * padding
    light_type = str(thumbnail["light_type"])
    bpy.ops.object.light_add(type=light_type, location=(0, -distance * 0.6, distance * 0.8))
    bpy.context.object.data.energy = float(thumbnail["light_energy"])
    if hasattr(bpy.context.object.data, "size"):
        bpy.context.object.data.size = float(thumbnail["light_size"])
    camera_location = (
        math.sin(yaw) * distance,
        -math.cos(yaw) * distance,
        math.sin(pitch) * distance,
    )
    bpy.ops.object.camera_add(location=camera_location, rotation=(1.05, 0, yaw))
    bpy.context.scene.camera = bpy.context.object


def render_thumbnail(path: Path, thumbnail: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    bpy.context.scene.render.resolution_x = int(thumbnail["width"])
    bpy.context.scene.render.resolution_y = int(thumbnail["height"])
    bpy.context.scene.world.color = hex_to_rgba(str(thumbnail["background"]))[:3]
    bpy.context.scene.render.filepath = str(path)
    bpy.ops.render.render(write_still=True)


def export_glb(path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    bpy.ops.export_scene.gltf(filepath=str(path), export_format="GLB")


def process_job(job: dict, thumbnail: dict) -> dict:
    try:
        clear_scene()
        import_model(Path(job["source_path"]))
        frame_camera(thumbnail)
        render_thumbnail(Path(job["thumbnail_path"]), thumbnail)
        if job.get("export_glb"):
            export_glb(Path(job["preview_path"]))
        return {"ok": True}
    except Exception as exc:
        return {"ok": False, "error": str(exc)}


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--jobs", required=True)
    parser.add_argument("--results", required=True)
    args = parser.parse_args()
    payload = json.loads(Path(args.jobs).read_text(encoding="utf-8"))
    results = {
        job["id"]: process_job(job, payload["thumbnail"])
        for job in payload["jobs"]
    }
    Path(args.results).write_text(json.dumps(results, ensure_ascii=False, indent=2), encoding="utf-8")
    return 0

function uriScheme(value) {
  const text = String(value || "");
  if (/^[a-zA-Z]:[\\/]/.test(text)) return "";
  const match = text.match(/^([a-zA-Z][a-zA-Z0-9+.-]*):/);
  return match ? `${match[1].toLowerCase()}:` : "";
}

function assertLocalUri(value) {
  const scheme = uriScheme(value);
  if (!scheme || scheme === "app:" || scheme === "data:") return;
  const blockedSchemes = new Set(["http:", "https:"]);
  if (blockedSchemes.has(scheme) || scheme) {
    throw new Error(`Blocked non-local asset URI: ${value}`);
  }
}

function normalizeLocalPath(value) {
  assertLocalUri(value);
  return String(value).replace(/\\/g, "/");
}

function encodeAssetPath(relativePath) {
  return relativePath.split("/").filter(Boolean).map(encodeURIComponent).join("/");
}

function toStudyRelative(filePath) {
  const root = normalizeLocalPath(state.runtime.studyRoot).replace(/\/+$/, "");
  const target = normalizeLocalPath(filePath);
  if (target.toLowerCase() === root.toLowerCase()) return "";
  if (!target.toLowerCase().startsWith(`${root.toLowerCase()}/`)) {
    throw new Error(`Path outside study-root: ${filePath}`);
  }
  return target.slice(root.length + 1);
}

function parentDir(filePath) {
  const normalized = normalizeLocalPath(filePath);
  const index = normalized.lastIndexOf("/");
  return index >= 0 ? normalized.slice(0, index) : "";
}

function joinLocalPath(baseDir, relativePath) {
  const parts = [];
  for (const part of `${normalizeLocalPath(baseDir)}/${String(relativePath).replace(/\\/g, "/")}`.split("/")) {
    if (!part || part === ".") continue;
    if (part === "..") parts.pop();
    else parts.push(part);
  }
  return parts.join("/");
}

function assetUrl(filePath) {
  assertLocalUri(filePath);
  return `app://asset/${encodeAssetPath(toStudyRelative(filePath))}`;
}

function resolveAssetReference(url, baseDir) {
  assertLocalUri(url);
  if (url.startsWith("app://") || url.startsWith("data:")) return url;
  if (/^[a-zA-Z]:[\\/]/.test(url) || url.startsWith("/")) return assetUrl(url);
  return assetUrl(joinLocalPath(baseDir, url));
}

function fitCamera(object) {
  const box = new THREE.Box3().setFromObject(object);
  const size = box.getSize(new THREE.Vector3()).length() || 1;
  const center = box.getCenter(new THREE.Vector3());
  controls.target.copy(center);

  camera.near = size / 100;
  camera.far = size * 100;

}

function fillFilter(select, values) {
  const current = select.value;
  const first = select.firstElementChild;
  select.innerHTML = "";
  select.appendChild(first);
  for (const value of [...new Set(values.filter(Boolean))].sort()) {
    const option = document.createElement("option");
    option.value = value;
    option.textContent = value;
    select.appendChild(option);

function applyFilters() {
  const query = document.getElementById("search").value.toLowerCase();
  const items = state.items.filter((item) => {
    const matchesQuery = item.name.toLowerCase().includes(query);
    const matchesFormat = !filterFormat.value || item.source_format === filterFormat.value;
    const matchesTexture = !filterTexture.value || item.texture_status === filterTexture.value;
    const matchesPreview = !filterPreview.value || item.preview_status === filterPreview.value;
    const matchesTags = !filterTags.value || (item.tags || []).includes(filterTags.value);
    return matchesQuery && matchesFormat && matchesTexture && matchesPreview && matchesTags;
  });
  state.filteredItems = items;
  renderStrip(items);
  status.textContent = `${items.length} / ${state.items.length}`;
}

function moveSelection(delta) {

  const index = Math.max(0, list.findIndex((item) => item.id === state.selected?.id));
  const next = list[(index + delta + list.length) % list.length];
  selectItem(next);
}

async function loadManifest() {
  state.runtime = await window.modelViewerBridge.getRuntime();
  const result = await window.modelViewerBridge.readManifest();
  if (!result.ok) {
    status.textContent = result.message;
    return;
  }
  state.manifest = result.manifest;
  state.items = state.manifest.items || [];
  fillFilter(filterFormat, state.items.map((item) => item.source_format));
  fillFilter(filterTexture, state.items.map((item) => item.texture_status));
  fillFilter(filterPreview, state.items.map((item) => item.preview_status));
  fillFilter(filterTags, state.items.flatMap((item) => item.tags || []));
  if (!state.items.length) {
    status.textContent = "本项目没有可预览模型";
    renderStrip([]);
    return;
  }
  applyFilters();
  if (state.items[0]) await selectItem(state.items[0]);
}

param(
  [string]$SkillRoot = "",
  [string]$ToolVersion = "1.0.0"
)

$ErrorActionPreference = "Stop"
$env:PYTHONIOENCODING = "utf-8"
[Console]::OutputEncoding = [Text.Encoding]::UTF8
$OutputEncoding = [Text.Encoding]::UTF8
chcp 65001 > $null

if ([string]::IsNullOrWhiteSpace($SkillRoot)) {
  $SkillRoot = Split-Path -Parent $PSScriptRoot
}

$sourceDir = Join-Path $SkillRoot "assets\model-gallery-viewer-src"
$workDir = Join-Path $SkillRoot "4.临时目录\工具构建\GameModelViewer"
$targetDir = Join-Path $SkillRoot "built-in-tools\GameModelViewer_$ToolVersion"

if (-not (Test-Path -LiteralPath $sourceDir)) {
  throw "Cannot find model viewer source: $sourceDir"
}

function Invoke-Npm {
  param([Parameter(Mandatory = $true)][string[]]$Arguments)
  $npmCommand = Get-Command "npm.cmd" -ErrorAction SilentlyContinue
  if (-not $npmCommand) { $npmCommand = Get-Command "npm" -ErrorAction Stop }
  & $npmCommand.Source @Arguments
}

New-Item -ItemType Directory -Force -Path $workDir | Out-Null
Copy-Item -LiteralPath (Join-Path $sourceDir "*") -Destination $workDir -Recurse -Force

Push-Location $workDir
try {
  Invoke-Npm @("install")
  Invoke-Npm @("run", "dist")
}
finally {
  Pop-Location
}

$portable = Get-ChildItem -LiteralPath (Join-Path $workDir "dist") -Filter "GameModelViewer.exe" -File |
  Sort-Object LastWriteTime -Descending |
  Select-Object -First 1
if (-not $portable) {
  throw "Electron build finished but GameModelViewer.exe was not found."
}

New-Item -ItemType Directory -Force -Path $targetDir | Out-Null
Copy-Item -LiteralPath $portable.FullName -Destination (Join-Path $targetDir "GameModelViewer.exe") -Force
New-Item -ItemType Directory -Force -Path (Join-Path $targetDir "LICENSES") | Out-Null
Set-Content -LiteralPath (Join-Path $targetDir "VERSION.txt") -Value "GameModelViewer $ToolVersion" -Encoding UTF8
Push-Location $workDir
try {
  $licenseJson = & (Get-Command "npm.cmd").Source @("ls", "--json", "--long")
  Set-Content -LiteralPath (Join-Path $targetDir "LICENSES\THIRD_PARTY_LICENSES.json") -Value $licenseJson -Encoding UTF8
}
finally {
  Pop-Location
}
Write-Output (Join-Path $targetDir "GameModelViewer.exe")
```
