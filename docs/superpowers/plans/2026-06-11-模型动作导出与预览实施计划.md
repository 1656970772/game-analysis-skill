## 架构审查

### 通用架构

动作导出不写成某个游戏的特殊规则，而抽象为四个通用对象：

- `AnimationAsset`：从 UnityPy / AssetRipper / 已有 glTF 中发现的动作资源。
- `ModelAnimationBinding`：动作与模型之间的关联证据、评分和状态。
- `ModelActionSidecar`：每个模型目录下的 `animations/actions.json`。
- `ViewerActionRecord`：写入模型预览 manifest，供 GameModelViewer 只读展示和播放。

数据流固定为：

```text
UnityPy all_resources_index.csv + metadata/*.json
  -> scripts/build_model_action_index.py
  -> 4.临时目录/中间索引/模型动作/{catalog.json,bindings.json,errors.jsonl}
  -> 1.分类资源/模型/<模型名>/animations/actions.json
  -> scripts/build_model_gallery_manifest.py
  -> 4.临时目录/中间索引/模型预览/manifest.json
  -> GameModelViewer 动作面板与 AnimationMixer 播放
```

### 设计模式适配

- Strategy：`scripts/model_action_common.py` 中定义动作来源策略，先实现 UnityPy metadata 策略，后续可加 AssetRipper `.anim`、FBX/glTF embedded animation 策略。
- Chain of Responsibility：动作与模型关联按证据链累加评分，证据包括同源容器、AnimatorController 引用、Avatar/骨骼线索、命名 token、已有 glTF animations。
- Adapter：viewer 新增 manifest adapter，把 schema v1 的无动作条目和 schema v2 的动作条目统一成内部 `actions` 数组。
- State Model：viewer 动作播放状态为 `idle | loading | playing | paused | unavailable | error`，不可播放动作不触发播放器。
- Configuration Object：目录名、权重、阈值、过滤规则、viewer 默认行为全部来自 `references/model-viewer-gallery-config.json`。

### 配置化覆盖范围

必须配置化：

- `animations.model_animation_dir`，默认 `animations`。
- `animations.model_actions_file`，默认 `actions.json`。

def merge_model_actions(existing: list[ModelAction], incoming: list[ModelAction]) -> list[ModelAction]:
    by_id = {action.id: action for action in existing}
    for action in incoming:
        previous = by_id.get(action.id)
        if previous is None or (action.playable and not previous.playable):
            by_id[action.id] = action
    return sorted(by_id.values(), key=lambda action: (not action.playable, action.name.lower(), action.id))

class BindingEvidenceScorer:
    reason: str

    def score(self, model: ModelCandidate, animation: AnimationCandidate, config: ActionConfig) -> float:
        raise NotImplementedError


class SameSourceFileScorer(BindingEvidenceScorer):
    reason = "same_source_file"

    def score(self, model: ModelCandidate, animation: AnimationCandidate, config: ActionConfig) -> float:
        if animation.source_file and animation.source_file == model.source_file:
            return config.association_weights.get(self.reason, 0.0)
        return 0.0


class SameContainerScorer(BindingEvidenceScorer):
    reason = "same_container"

    def score(self, model: ModelCandidate, animation: AnimationCandidate, config: ActionConfig) -> float:
        if model.source_container and model.source_container == animation.source_container:
            return config.association_weights.get(self.reason, 0.0)
        return 0.0


class AnimatorControllerReferenceScorer(BindingEvidenceScorer):
    reason = "animator_controller_reference"

    def score(self, model: ModelCandidate, animation: AnimationCandidate, config: ActionConfig) -> float:
        if model.id in animation.referenced_model_ids or model.name in animation.referenced_model_names:
            return config.association_weights.get(self.reason, 0.0)
        return 0.0


class NameSimilarityScorer(BindingEvidenceScorer):
    reason = "name_similarity"

    def score(self, model: ModelCandidate, animation: AnimationCandidate, config: ActionConfig) -> float:
        similarity = token_similarity(model.name, animation.name)
        return min(similarity, 1.0) * config.association_weights.get(self.reason, 0.0)

def read_csv_utf8(path: Path) -> list[dict[str, str]]:
    with path.open(encoding="utf-8-sig", newline="") as handle:

def token_similarity(left: str, right: str) -> float:
    left_tokens = set(re.findall(r"[a-z0-9]+", left.lower()))
    right_tokens = set(re.findall(r"[a-z0-9]+", right.lower()))
    if not left_tokens or not right_tokens:
        return 0.0
    return len(left_tokens & right_tokens) / len(left_tokens | right_tokens)


def normalized_source_container(source_file: str) -> str:
    normalized = source_file.replace("\\", "/").strip().lower()
    if not normalized:

    parent = normalized.rsplit("/", 1)[0] if "/" in normalized else normalized
    return parent

def bind_actions(
    model: ModelCandidate,
    animations: list[AnimationCandidate],
    config: ActionConfig,
    scorers: tuple[BindingEvidenceScorer, ...] = DEFAULT_SCORERS,
) -> list[ModelAction]:
    actions: list[ModelAction] = []
    for animation in animations:
        score = 0.0
        reasons: list[str] = []
        for scorer in scorers:
            contribution = scorer.score(model, animation, config)
            if contribution > 0:
                score += contribution
                reasons.append(scorer.reason)
        if score < config.association_threshold:
            continue
        reason_code = "missing_transform_bindings"
        if reason_code not in config.unplayable_reason_codes:
            reason_code = config.unplayable_reason_codes[0] if config.unplayable_reason_codes else "metadata_only"
        actions.append(
            ModelAction(
                id=action_id("AnimationClip", animation.source_file, animation.source_path_id, animation.name),
                name=animation.name,
                source_type="AnimationClip",
                source_file=animation.source_file,
                source_path_id=str(animation.source_path_id),
                playable=False,
                duration_seconds=animation.duration_seconds,
                loop=animation.loop,
                binding_score=round(score, 4),
                binding_reasons=reasons,
                metadata={"sample_rate": animation.sample_rate, "has_muscle_clip": animation.duration_seconds > 0},
                unavailable_reason={
                    "code": reason_code,
                    "message": "Animation metadata was exported, but glTF transform tracks were not reconstructed.",
                },
            )
        )
    return actions


def build_model_action_index(
    study_root: Path,
    config_document: dict[str, Any] | None = None,
    source_strategies: tuple[AnimationSourceStrategy, ...] = (UnityPyMetadataAnimationStrategy(),),
) -> dict[str, Any]:
    study_root = Path(study_root).resolve(strict=False)
    if config_document is None:
        config_document = load_model_viewer_config()
    action_config = action_config_from_model_viewer_config(config_document)
    index_dir = study_root / action_config.index_dir
    index_dir.mkdir(parents=True, exist_ok=True)
    models = build_model_candidates(study_root, action_config)
    animations = []
    for strategy in source_strategies:
        animations.extend(strategy.collect(study_root, action_config))
    models_with_actions = 0
    action_count = 0
    catalog = {"animations": [animation.__dict__ for animation in animations]}
    bindings: list[dict[str, Any]] = []
    for model in models:
        actions = bind_actions(model, animations, action_config)
        if not actions:
            continue
        models_with_actions += 1
        action_count += len(actions)
        write_model_actions(model.model_dir, model.id, model.name, actions, action_config)
        bindings.append({"model_id": model.id, "model_name": model.name, "action_count": len(actions)})
    write_json_utf8(index_dir / "catalog.json", catalog)
    write_json_utf8(index_dir / "bindings.json", {"bindings": bindings})
    summary = {
        "study_root": str(study_root),
        "model_count": len(models),
        "animation_count": len(animations),
        "models_with_actions": models_with_actions,
        "action_count": action_count,
        "index_dir": str(index_dir),
    }

```python
def collect_embedded_clips_from_gltf(model_file: Path) -> list[dict[str, Any]]:
    if model_file.suffix.lower() != ".gltf":
        return []
    try:
        payload = json.loads(model_file.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeDecodeError, json.JSONDecodeError):
        return []
    clips: list[dict[str, Any]] = []
    accessors = payload.get("accessors", [])
    for index, animation in enumerate(payload.get("animations", [])):
        channels = animation.get("channels") or []
        samplers = animation.get("samplers") or []
        if not channels or not samplers:
            continue
        duration = 0.0
        for sampler in samplers:
            input_index = sampler.get("input")
            if isinstance(input_index, int) and input_index < len(accessors):
                maximum = accessors[input_index].get("max") or []
                if maximum:
                    duration = max(duration, float(maximum[0] or 0.0))
        name = animation.get("name") or f"Clip {index}"
        clips.append(
            {
                "id": f"embedded-{index}-{name}",
                "name": name,
                "clip_index": index,
                "duration_seconds": duration,
                "loop": True,
                "relative_clip_path": f"animations/{index}",
            }
        )
    return clips
```

  stats.playable_animation_count = actions.filter((action) => action.playable).length;
  return {
    ...item,
    actions,
    stats
  };
}

export function sortActions(actions, sortOrder = ["playable", "name"]) {
  const order = Array.isArray(sortOrder) ? sortOrder : ["playable", "name"];
  return [...actions].sort((left, right) => {
    for (const key of order) {
      if (key === "playable" && left.playable !== right.playable) {
        return left.playable ? -1 : 1;
      }
      if (key === "name") {
        const byName = left.name.localeCompare(right.name, "zh-Hans-CN");
        if (byName !== 0) return byName;
      }
    }
    return left.id.localeCompare(right.id);
  });
}

function adaptAction(action) {
  return {
    id: action.id || action.name,
    name: action.name || action.id || "未命名动作",
    playable: action.playable === true,
    clipIndex: Number.isInteger(action.clip_index) ? action.clip_index : null,
    relativeClipPath: action.relative_clip_path || "",
    durationSeconds: Number(action.duration_seconds || 0),
    loop: action.loop !== false,
    unavailableReason: action.unavailable_reason || null
  };
}

import * as THREE from "three";

export class AnimationPlayer {
  constructor() {
    this.mixer = null;
    this.clips = [];
    this.activeAction = null;
    this.state = { status: "idle", activeClipName: "", error: "" };
  }

  load(root, clips) {
    this.dispose();
    this.mixer = root ? new THREE.AnimationMixer(root) : null;
    this.clips = Array.isArray(clips) ? clips : [];
    this.state = { status: "idle", activeClipName: "", error: "" };
    return this.state;
  }

  playAction(actionRecord, { loop = true, speed = 1 } = {}) {
    if (!actionRecord || actionRecord.playable !== true) {
      this.state = {
        status: "unavailable",
        activeClipName: "",
        error: actionRecord?.unavailableReason?.message || "该动作不可播放。"
      };
      return this.state;
    }
    const clip = this.findClip(actionRecord);
    if (!clip || !this.mixer) {
      this.state = { status: "unavailable", activeClipName: "", error: `动作轨道不存在：${actionRecord.name}` };
      return this.state;
    }
    this.mixer.stopAllAction();
    this.activeAction = this.mixer.clipAction(clip);
    this.activeAction.reset();
    this.activeAction.setLoop(loop ? THREE.LoopRepeat : THREE.LoopOnce);
    this.activeAction.timeScale = Number(speed || 1);
    this.activeAction.play();
    this.state = { status: "playing", activeClipName: actionRecord.name, error: "" };
    return this.state;
  }

  findClip(actionRecord) {
    if (Number.isInteger(actionRecord.clipIndex) && this.clips[actionRecord.clipIndex]) {
      return this.clips[actionRecord.clipIndex];
    }
    if (actionRecord.relativeClipPath) {
      const byPath = this.clips.find((candidate) =>
        candidate.userData?.relativeClipPath === actionRecord.relativeClipPath ||
        candidate.userData?.sourcePath === actionRecord.relativeClipPath
      );
      if (byPath) return byPath;
    }
    return this.clips.find((candidate) => candidate.name === actionRecord.name);
  }

  pause() {
    if (this.activeAction) this.activeAction.paused = true;
    this.state = { ...this.state, status: "paused" };
    return this.state;
  }

  stop() {
    if (this.activeAction) this.activeAction.stop();
    this.activeAction = null;
    this.state = { status: "idle", activeClipName: "", error: "" };
    return this.state;
  }

  update(deltaSeconds) {
    if (this.mixer && this.state.status === "playing") this.mixer.update(deltaSeconds);
  }

  dispose() {
    if (this.mixer) this.mixer.stopAllAction();
    this.mixer = null;
    this.clips = [];
    this.activeAction = null;
  }
}

```python
from __future__ import annotations

import argparse
import json
from pathlib import Path
from typing import Any

from scripts.model_action_common import action_config_from_model_viewer_config
from scripts.model_gallery_common import load_model_viewer_config


def configured_path(root: Path, config: dict[str, Any], key: str) -> Path:
    relative = config.get("paths", {}).get(key, key)
    return root / relative


def summarize_model_actions(study_root: Path, config_document: dict[str, Any] | None = None) -> dict[str, Any]:
    study_root = Path(study_root).resolve(strict=False)
    config = config_document or load_model_viewer_config()
    action_config = action_config_from_model_viewer_config(config)
    models_dir = configured_path(study_root, config, "models_dir")
    configured_formats = action_config.verify_model_formats or tuple(config.get("formats", {}).get("source_priority", ["gltf", "glb", "fbx"]))
    suffixes = {"." + suffix.lower().lstrip(".") for suffix in configured_formats}
    model_files = [path for path in models_dir.rglob("*") if path.is_file() and path.suffix.lower() in suffixes] if models_dir.exists() else []
    action_files = [
        path
        for path in models_dir.rglob(action_config.model_actions_file)
        if path.parent.name == action_config.model_animation_dir
    ] if models_dir.exists() else []
    action_count = 0
    playable = 0
    unplayable = 0
    missing_reason = 0
    manifest_action_mismatch_count = 0
    for action_file in action_files:
        payload = json.loads(action_file.read_text(encoding="utf-8-sig"))
        for action in payload.get("actions", []):
            action_count += 1
            if action.get("playable") is True:
                playable += 1
            else:
                unplayable += 1
                reason = action.get("unavailable_reason")
                if not isinstance(reason, dict) or not reason.get("code") or not reason.get("message"):
                    missing_reason += 1
    manifest_path = study_root / config.get("paths", {}).get("preview_index_dir", "4.临时目录/中间索引/模型预览") / config.get("paths", {}).get("manifest_name", "manifest.json")
    if manifest_path.exists():
        manifest = json.loads(manifest_path.read_text(encoding="utf-8-sig"))
        for item in manifest.get("items", []):
            model_dir = Path(item.get("model_dir") or "")
            sidecar = model_dir / action_config.model_animation_dir / action_config.model_actions_file
            sidecar_count = 0
            sidecar_playable_count = 0
            if sidecar.exists():
                payload = json.loads(sidecar.read_text(encoding="utf-8-sig"))
                sidecar_count = len(payload.get("actions", []))
                sidecar_playable_count = sum(1 for action in payload.get("actions", []) if action.get("playable") is True)
            manifest_actions = item.get("actions", [])
            manifest_playable_count = sum(1 for action in manifest_actions if action.get("playable") is True)
            if len(manifest_actions) != sidecar_count or manifest_playable_count != sidecar_playable_count:
                manifest_action_mismatch_count += 1
    return {
        "study_root": str(study_root),
        "model_count": len(model_files),
        "models_with_actions": len(action_files),
        "action_count": action_count,
        "playable_action_count": playable,
        "unplayable_action_count": unplayable,
        "missing_unavailable_reason_count": missing_reason,
        "manifest_action_mismatch_count": manifest_action_mismatch_count,
    }


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Verify exported model actions.")
    parser.add_argument("--study-root", required=True, type=Path)
    parser.add_argument("--config", type=Path, default=None)
    args = parser.parse_args(argv)
    config = load_model_viewer_config(args.config) if args.config else load_model_viewer_config()
    print(json.dumps(summarize_model_actions(args.study_root, config), ensure_ascii=False, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
